🔌 مركز الربط والتكامل البرمجي مع أودو (Odoo REST & XML-RPC)

جميع أوامر التشغيل وعروض الأسعار والعملاء مهيأة بحمولات JSON مطابقة لهياكل أودو 17 / 18.

OpenAPI 3.0 Ready

كود بايثون للربط المباشر مع أودو (XML-RPC): 

# كود ربط أودو مع تطبيق مصنع الستائر
import xmlrpc.client
import urllib.request
import json

ODOO_URL = 'http://localhost:8069'
ODOO_DB = 'curtain_factory_db'
ODOO_USER = 'admin'
ODOO_PASSWORD = 'admin_password'

CURTAIN_APP_URL = 'http://localhost:3000'

# 1. جلب بيانات أمر التشغيل المهيأ لأودو
req = urllib.request.urlopen(f"{CURTAIN_APP_URL}/api/v1/odoo/payload/job/1")
job_payload = json.loads(req.read().decode('utf-8'))['payload']

# 2. تسجيل الدخول في أودو
common = xmlrpc.client.ServerProxy(f'{ODOO_URL}/xmlrpc/2/common')
uid = common.authenticate(ODOO_DB, ODOO_USER, ODOO_PASSWORD, {})
models = xmlrpc.client.ServerProxy(f'{ODOO_URL}/xmlrpc/2/object')

# 3. إنشاء السجل داخل موديول curtain.job
job_id = models.execute_kw(
    ODOO_DB, uid, ODOO_PASSWORD,
    'curtain.job', 'create',
    [job_payload['values']]
)
print(f"تم ربط السجل بنجاح في أودو برقم: {job_id}")
